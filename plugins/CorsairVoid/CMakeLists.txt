cmake_minimum_required(VERSION 3.21)
project(OpenRGBCorsairVoidLighting VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(OPENRGB_SOURCE_DIR "" CACHE PATH "Complete OpenRGB 1.1 source tree")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MSVC x64 is required for this OpenRGB build")
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets)
add_library(OpenRGBCorsairVoidLighting MODULE VoidPlugin.cpp VoidPlugin.h VoidHid.cpp VoidHid.h plugin.json.in)
target_include_directories(OpenRGBCorsairVoidLighting PRIVATE "${OPENRGB_SOURCE_DIR}" "${OPENRGB_SOURCE_DIR}/RGBController" "${OPENRGB_SOURCE_DIR}/dependencies/json" "${OPENRGB_SOURCE_DIR}/dependencies/hidapi-hotplug-win/include")
target_link_libraries(OpenRGBCorsairVoidLighting PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_definitions(OpenRGBCorsairVoidLighting PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(OpenRGBCorsairVoidLighting PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
if(ASUS_LIGHTING_BUILD_CHECKS)
add_executable(VoidChecks VoidChecks.cpp)
get_target_property(VOID_INCLUDES OpenRGBCorsairVoidLighting INCLUDE_DIRECTORIES)
target_include_directories(VoidChecks PRIVATE ${VOID_INCLUDES})
target_link_libraries(VoidChecks PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_definitions(VoidChecks PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(VoidChecks PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/checks")
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/PluginMetadata.cmake")
configure_lighting_metadata(OpenRGBCorsairVoidLighting "${CMAKE_CURRENT_SOURCE_DIR}/plugin.json.in" "VoidPluginMetadata.json" "https://github.com/TheCrimsonborn/OpenRGB-ASUS-Lighting/tree/main/plugins/CorsairVoid")
